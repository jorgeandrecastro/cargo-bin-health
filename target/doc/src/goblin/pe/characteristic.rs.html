79/// Image only, Windows CE, and Microsoft Windows NT and later. This indicates that the file does not
80/// contain base relocations and must therefore be loaded at its preferred base address. If the base address
81/// is not available, the loader reports an error. The default behavior of the linker is to strip base relocations
82/// from executable (EXE) files.
83pub const IMAGE_FILE_RELOCS_STRIPPED: u16 = 0x0001;
84
85/// Image only. This indicates that the image file is valid and can be run.
86/// If this flag is not set, it indicates a linker error.
87pub const IMAGE_FILE_EXECUTABLE_IMAGE: u16 = 0x0002;
88
89/// COFF line numbers have been removed. This flag is deprecated and should be zero.
90pub const IMAGE_FILE_LINE_NUMS_STRIPPED: u16 = 0x0004;
91
92/// COFF symbol table entries for local symbols have been removed. This flag is deprecated and should be zero.
93pub const IMAGE_FILE_LOCAL_SYMS_STRIPPED: u16 = 0x0008;
94
95/// Obsolete. Aggressively trim working set. This flag is deprecated for Windows 2000 and later and must be zero.
96pub const IMAGE_FILE_AGGRESSIVE_WS_TRIM: u16 = 0x0010;
97
98/// Application can handle > 2-GB addresses.
99pub const IMAGE_FILE_LARGE_ADDRESS_AWARE: u16 = 0x0020;
100
101/// This flag is reserved for future use.
102pub const RESERVED: u16 = 0x0040;
103
104/// Little endian: the least significant bit (LSB) precedes the most significant bit (MSB) in memory.
105/// This flag is deprecated and should be zero.
106pub const IMAGE_FILE_BYTES_REVERSED_LO: u16 = 0x0080;
107
108/// Machine is based on a 32-bit-word architecture.
109pub const IMAGE_FILE_32BIT_MACHINE: u16 = 0x0100;
110
111/// Debugging information is removed from the image file.
112pub const IMAGE_FILE_DEBUG_STRIPPED: u16 = 0x0200;
113
114/// If the image is on removable media, fully load it and copy it to the swap file.
115pub const IMAGE_FILE_REMOVABLE_RUN_FROM_SWAP: u16 = 0x0400;
116
117/// If the image is on network media, fully load it and copy it to the swap file.
118pub const IMAGE_FILE_NET_RUN_FROM_SWAP: u16 = 0x0800;
119
120/// The image file is a system file, not a user program.
121pub const IMAGE_FILE_SYSTEM: u16 = 0x1000;
122
123/// The image file is a dynamic-link library (DLL). Such files are considered executable files for almost all purposes, although they cannot be directly run.
124pub const IMAGE_FILE_DLL: u16 = 0x2000;
125
126/// The file should be run only on a uniprocessor machine.
127pub const IMAGE_FILE_UP_SYSTEM_ONLY: u16 = 0x4000;
128
129/// Big endian: the MSB precedes the LSB in memory. This flag is deprecated and should be zero.
130pub const IMAGE_FILE_BYTES_REVERSED_HI: u16 = 0x8000;
131
132/// Checks whether the characteristics value indicates that the file is a DLL (dynamically-linked library).
133pub fn is_dll(characteristics: u16) -> bool {
134    characteristics & IMAGE_FILE_DLL == IMAGE_FILE_DLL
135}
136
137/// Checks whether the characteristics value indicates that the file is an executable.
138pub fn is_exe(characteristics: u16) -> bool {
139    characteristics & IMAGE_FILE_EXECUTABLE_IMAGE == IMAGE_FILE_EXECUTABLE_IMAGE
140}
