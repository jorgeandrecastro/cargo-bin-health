1//! Constants for characteristics of image files. These constants are used in the
2//! [`goblin::pe::optional_header::WindowsFields::dll_characteristics`](crate::pe::optional_header::WindowsFields::dll_characteristics)
3//! field.
4//!
5//! The values 0x0001, 0x0002, 0x0004, 0x0008 are reserved for future use and must be zero.
6
7/// Image can handle a high entropy 64-bit virtual address space.
8pub const IMAGE_DLLCHARACTERISTICS_HIGH_ENTROPY_VA: u16 = 0x0020;
9
10/// DLL can be relocated at load time.
11pub const IMAGE_DLLCHARACTERISTICS_DYNAMIC_BASE: u16 = 0x0040;
12
13/// Code Integrity checks are enforced.
14pub const IMAGE_DLLCHARACTERISTICS_FORCE_INTEGRITY: u16 = 0x0080;
15
16/// Image is NX compatible.
17pub const IMAGE_DLLCHARACTERISTICS_NX_COMPAT: u16 = 0x0100;
18
19/// Isolation aware, but do not isolate the image.
20pub const IMAGE_DLLCHARACTERISTICS_NO_ISOLATION: u16 = 0x0200;
21
22/// Does not use structured exception (SE) handling. No SE handler may be called in this image.
23pub const IMAGE_DLLCHARACTERISTICS_NO_SEH: u16 = 0x0400;
24
25/// Do not bind the image.
26pub const IMAGE_DLLCHARACTERISTICS_NO_BIND: u16 = 0x0800;
27
28/// Image must execute in an AppContainer.
29pub const IMAGE_DLLCHARACTERISTICS_APPCONTAINER: u16 = 0x1000;
30
31/// A WDM driver.
32pub const IMAGE_DLLCHARACTERISTICS_WDM_DRIVER: u16 = 0x2000;
33
34/// Image supports Control Flow Guard.
35pub const IMAGE_DLLCHARACTERISTICS_GUARD_CF: u16 = 0x4000;
36
37/// Terminal Server aware.
38pub const IMAGE_DLLCHARACTERISTICS_TERMINAL_SERVER_AWARE: u16 = 0x8000;
